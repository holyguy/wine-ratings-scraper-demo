  base_url: https://www.wine-searcher.com
  listing_selector: ".wine-card"
  name: ".wine-card__name"
  rating: ".wine-card__rating span"
  price: ".wine-card__price .amount"
  vintage: ".wine-card__vintage"
  pagination: ".pagination__next a"
